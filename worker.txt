function escapeHtml(text) {

    return String(text)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}

const COOKIE_NAME =
    "papamono_session";

const SESSION_MAX_AGE =
    5 * 60 * 1000;


// =====================================================
// UTILIDADES CRYPTO
// =====================================================

function bytesToHex(bytes) {

    return Array.from(
        bytes,
        function(b) {
            return b
                .toString(16)
                .padStart(2, "0");
        }
    ).join("");
}


function hexToBytes(hex) {

    const bytes =
        new Uint8Array(
            hex.length / 2
        );

    for (
        let i = 0;
        i < bytes.length;
        i++
    ) {

        bytes[i] =
            parseInt(
                hex.substr(i * 2, 2),
                16
            );
    }

    return bytes;
}


async function getKey(secret) {

    return crypto.subtle.importKey(
        "raw",
        new TextEncoder().encode(secret),
        {
            name: "HMAC",
            hash: "SHA-256"
        },
        false,
        [
            "sign",
            "verify"
        ]
    );
}


async function createSession(secret) {

    const random =
        crypto.getRandomValues(
            new Uint8Array(32)
        );

    const timestamp =
        Date.now().toString();

    const randomHex =
        bytesToHex(random);

    const value =
        timestamp +
        "." +
        randomHex;

    const key =
        await getKey(secret);

    const signature =
        await crypto.subtle.sign(
            "HMAC",
            key,
            new TextEncoder().encode(value)
        );

    return (
        value +
        "." +
        bytesToHex(
            new Uint8Array(signature)
        )
    );
}


async function validSession(
    session,
    secret
) {

    if (!session) {
        return false;
    }

    const parts =
        session.split(".");

    if (parts.length !== 3) {
        return false;
    }

    const timestamp =
        Number(parts[0]);

    const random =
        parts[1];

    const signature =
        parts[2];

    if (
        !Number.isFinite(timestamp) ||
        random.length !== 64 ||
        signature.length !== 64
    ) {
        return false;
    }

    if (
        Date.now() - timestamp >
        SESSION_MAX_AGE
    ) {
        return false;
    }

    try {

        const value =
            timestamp +
            "." +
            random;

        const key =
            await getKey(secret);

        return await crypto.subtle.verify(
            "HMAC",
            key,
            hexToBytes(signature),
            new TextEncoder().encode(value)
        );

    } catch {

        return false;
    }
}


function getSession(request) {

    const cookie =
        request.headers.get("Cookie") || "";

    for (
        const part of cookie.split(";")
    ) {

        const trimmed =
            part.trim();

        if (
            trimmed.startsWith(
                COOKIE_NAME + "="
            )
        ) {

            return trimmed.slice(
                COOKIE_NAME.length + 1
            );
        }
    }

    return null;
}


async function requireAdmin(
    request,
    env
) {

    if (!env.UPLOAD_PASSWORD) {
        return false;
    }

    return await validSession(
        getSession(request),
        env.UPLOAD_PASSWORD
    );
}

const MAX_LOGIN_ATTEMPTS = 5;
const LOGIN_WINDOW_SECONDS = 60;
const LOGIN_BLOCK_SECONDS = 600;


async function checkLoginRateLimit(
    request,
    env
) {

    if (!env.MY_DATA_KV) {
        return {
            allowed: true
        };
    }

    const ip =
        request.headers.get(
            "CF-Connecting-IP"
        ) || "unknown";

    const key =
        "admin_rl:" + ip;

    try {

        const current =
            await env.MY_DATA_KV.get(
                key,
                {
                    type: "json"
                }
            );

        const now =
            Date.now();

        if (
            current &&
            current.blockedUntil &&
            now < current.blockedUntil
        ) {

            return {
                allowed: false,

                retryAfter:
                    Math.ceil(
                        (
                            current.blockedUntil -
                            now
                        ) / 1000
                    )
            };
        }

        if (
            !current ||
            !current.startedAt ||
            now -
            current.startedAt >=
            LOGIN_WINDOW_SECONDS * 1000
        ) {

            await env.MY_DATA_KV.put(
                key,
                JSON.stringify({
                    attempts: 0,
                    startedAt: now
                }),
                {
                    expirationTtl:
                        LOGIN_WINDOW_SECONDS
                }
            );

            return {
                allowed: true
            };
        }

        if (
            current.attempts >=
            MAX_LOGIN_ATTEMPTS
        ) {

            const blockedUntil =
                now +
                LOGIN_BLOCK_SECONDS * 1000;

            await env.MY_DATA_KV.put(
                key,
                JSON.stringify({
                    attempts:
                        current.attempts,

                    startedAt:
                        current.startedAt,

                    blockedUntil:
                        blockedUntil
                }),
                {
                    expirationTtl:
                        LOGIN_BLOCK_SECONDS
                }
            );

            return {
                allowed: false,

                retryAfter:
                    LOGIN_BLOCK_SECONDS
            };
        }

        return {
            allowed: true
        };

    } catch {

        return {
            allowed: true
        };
    }
}


async function registerFailedLogin(
    request,
    env
) {

    if (!env.MY_DATA_KV) {
        return;
    }

    const ip =
        request.headers.get(
            "CF-Connecting-IP"
        ) || "unknown";

    const key =
        "admin_rl:" + ip;

    try {

        const current =
            await env.MY_DATA_KV.get(
                key,
                {
                    type: "json"
                }
            );

        const now =
            Date.now();

        let attempts = 1;
        let startedAt = now;

        if (
            current &&
            current.startedAt &&
            now -
            current.startedAt <
            LOGIN_WINDOW_SECONDS * 1000
        ) {

            attempts =
                (current.attempts || 0) + 1;

            startedAt =
                current.startedAt;
        }

        const data = {
            attempts:
                attempts,

            startedAt:
                startedAt
        };

        if (
            attempts >=
            MAX_LOGIN_ATTEMPTS
        ) {

            data.blockedUntil =
                now +
                LOGIN_BLOCK_SECONDS * 1000;

            await env.MY_DATA_KV.put(
                key,
                JSON.stringify(data),
                {
                    expirationTtl:
                        LOGIN_BLOCK_SECONDS
                }
            );

        } else {

            await env.MY_DATA_KV.put(
                key,
                JSON.stringify(data),
                {
                    expirationTtl:
                        LOGIN_WINDOW_SECONDS
                }
            );
        }

    } catch {}
}


async function clearLoginRateLimit(
    request,
    env
) {

    if (!env.MY_DATA_KV) {
        return;
    }

    const ip =
        request.headers.get(
            "CF-Connecting-IP"
        ) || "unknown";

    try {

        await env.MY_DATA_KV.delete(
            "admin_rl:" + ip
        );

    } catch {}
}

async function handleAdminLogin(
    request,
    env
) {

    if (!env.UPLOAD_PASSWORD) {

        return new Response(
            "UPLOAD_PASSWORD no está configurado.",
            {
                status: 500
            }
        );
    }

    const rate =
        await checkLoginRateLimit(
            request,
            env
        );

    if (!rate.allowed) {

        return new Response(
            LOGIN_HTML,
            {
                status: 429,

                headers: {
                    "Content-Type":
                        "text/html;charset=UTF-8",

                    "Retry-After":
                        String(
                            rate.retryAfter
                        )
                }
            }
        );
    }

    const form =
        await request.formData();

    const password =
        form.get("password");

    if (
        typeof password !== "string" ||
        password !== env.UPLOAD_PASSWORD
    ) {

        await registerFailedLogin(
            request,
            env
        );

        return new Response(
            LOGIN_ERROR_HTML,
            {
                status: 401,

                headers: {
                    "Content-Type":
                        "text/html;charset=UTF-8"
                }
            }
        );
    }

    await clearLoginRateLimit(
        request,
        env
    );

    const session =
        await createSession(
            env.UPLOAD_PASSWORD
        );

    return new Response(
        null,
        {
            status: 302,

            headers: {

                "Location":
                    "/admin",

                "Set-Cookie":
                    `${COOKIE_NAME}=${session}; ` +
                    `Path=/; ` +
                    `HttpOnly; ` +
                    `Secure; ` +
                    `SameSite=Strict; ` +
                    `Max-Age=300`
            }
        }
    );
}

async function handleFiles(env) {

    if (!env.PUBLIC_BUCKET) {

        return new Response(
            JSON.stringify({
                files: [],
                error:
                    "PUBLIC_BUCKET no está configurado."
            }),
            {
                status: 500,

                headers: {
                    "Content-Type":
                        "application/json",
                    "Cache-Control":
                        "no-store"
                }
            }
        );
    }

    try {

        const files = [];

        let cursor = undefined;

        do {

            const options = {
                limit: 1000
            };

            if (cursor) {
                options.cursor = cursor;
            }

            const listed =
                await env.PUBLIC_BUCKET.list(
                    options
                );

            for (
                const object of listed.objects
            ) {

                files.push({
                    key:
                        object.key,

                    size:
                        object.size
                });
            }

            cursor =
                listed.truncated
                    ? listed.cursor
                    : undefined;

        } while (cursor);


        files.sort(
            function(a, b) {

                return a.key.localeCompare(
                    b.key,
                    undefined,
                    {
                        numeric: true
                    }
                );
            }
        );


        return new Response(
            JSON.stringify({
                files:
                    files
            }),
            {
                status: 200,

                headers: {
                    "Content-Type":
                        "application/json",

                    "Cache-Control":
                        "no-store"
                }
            }
        );

    } catch (err) {

        console.error(
            "R2 LIST ERROR",
            err
        );

        return new Response(
            JSON.stringify({
                files: [],

                error:
                    err &&
                    err.message
                        ? err.message
                        : String(err)
            }),
            {
                status: 500,

// =====================================================
// R2 UPLOAD
// =====================================================

async function handleUpload(
    request,
    env
) {

    if (!env.PUBLIC_BUCKET) {

        return new Response(
            JSON.stringify({
                success: false,

                error:
                    "PUBLIC_BUCKET no está configurado."
            }),
            {
                status: 500,

                headers: {
                    "Content-Type":
                        "application/json"
                }
            }
        );
    }

    try {

        const formData =
            await request.formData();

        const file =
            formData.get("file");

        if (
            !file ||
            typeof file.stream !== "function"
        ) {

            return new Response(
                JSON.stringify({
                    success: false,

                    error:
                        "No se recibió archivo."
                }),
                {
                    status: 400,

                    headers: {
                        "Content-Type":
                            "application/json"
                    }
                }
            );
        }

        if (!file.name) {

            return new Response(
                JSON.stringify({
                    success: false,

                    error:
                        "El archivo no tiene nombre."
                }),
                {
                    status: 400,

                    headers: {
                        "Content-Type":
                            "application/json"
                    }
                }
            );
        }

        const key =
            file.name;

        await env.PUBLIC_BUCKET.put(
            key,
            file.stream(),
            {
                httpMetadata: {
                    contentType:
                        file.type ||
                        "application/octet-stream"
                }
            }
        );

        return new Response(
            JSON.stringify({
                success:
                    true,

                key:
                    key,

                url:
                    "/f/" +
                    encodeURIComponent(key)
            }),
            {
                status: 200,

                headers: {
                    "Content-Type":
                        "application/json",

                    "Cache-Control":
                        "no-store"
                }
            }
        );

    } catch (err) {

        console.error(
            "R2 UPLOAD ERROR",
            err
        );

        return new Response(
            JSON.stringify({
                success:
                    false,

                error:
                    err &&
                    err.message
                        ? err.message
                        : String(err)
            }),
            {
                status: 500,

                headers: {
                    "Content-Type":
                        "application/json"
                }
            }
        );
    }
}


// =====================================================
// R2 GET
// =====================================================

async function handleFile(
    request,
    env
) {

    if (!env.PUBLIC_BUCKET) {

        return new Response(
            "PUBLIC_BUCKET no está configurado.",
            {
                status: 500
            }
        );
    }

    const url =
        new URL(request.url);

    const encodedKey =
        url.pathname.slice(3);

    if (!encodedKey) {

        return new Response(
            "Archivo no especificado.",
            {
                status: 400
            }
        );
    }

    let key;

    try {

        key =
            decodeURIComponent(
                encodedKey
            );

    } catch {

        return new Response(
            "Nombre de archivo inválido.",
            {
                status: 400
            }
        );
    }

    try {

        const object =
            await env.PUBLIC_BUCKET.get(
                key
            );

        if (!object) {

            return new Response(
                "Archivo no encontrado.",
                {
                    status: 404
                }
            );
        }

        const headers =
            new Headers();

        object.writeHttpMetadata(
            headers
        );

        headers.set(
            "Content-Type",
            "text/plain; charset=utf-8"
        );

        headers.set(
            "Cache-Control",
            "public, max-age=0, must-revalidate"
        );

        return new Response(
            object.body,
            {
                status: 200,
                headers: headers
            }
        );

    } catch (err) {

        console.error(
            "R2 GET ERROR",
            err
        );

        return new Response(
            "Error leyendo archivo.",
            {
                status: 500
            }
        );
    }
}

    request,

) {

        return new Response(
            "PUBLIC_BUCKET no está configurado.",
            {
                status: 500
            }
        );
    }

    const url =
        new URL(request.url);

    const encodedKey =
        url.pathname.slice(

        );

    if (!encodedKey) {

        return new Response(
            "Archivo no especificado.",
            {
                status: 400
            }
        );
    }

    let key;

    try {

        key =
            decodeURIComponent(
                encodedKey
            );

        return new Response(

    ) {

            }
        );

        const object =
            await env.PUBLIC_BUCKET.get(
                key
            );

        if (!object) {

            return new Response(
                "Archivo no encontrado.",
                {
                    status: 404
                }
            );
        }

        const content =
            await object.text();

        return new Response(
            EDITOR_HTML(
                key,
                content
            ),
            {
                status: 200,

                headers: {
                    "Content-Type":
                        "text/html;charset=UTF-8",

                    "Cache-Control":
                        "private, no-store"
                }
            }
        );

    } catch (err) {

        console.error(
            "R2 EDIT ERROR",
            err
        );

        return new Response(
            "Error leyendo archivo.",

                status: 500
            }
        );

// =====================================================
// WORKER
// =====================================================

export default {

    async fetch(
        request,
        env
    ) {

        const url =
            new URL(request.url);


        // =================================================
        // LOGIN
        // =================================================

        if (
            url.pathname ===
                "/admin/login" &&
            request.method === "POST"
        ) {

            return handleAdminLogin(
                request,
                env
            );
        }


        // =================================================
        // LOGOUT
        // =================================================

        if (
            url.pathname ===
                "/admin/logout" &&
            request.method === "POST"
        ) {

            return handleAdminLogout();
        }


        // =================================================
        // ADMIN
        // =================================================

        if (
            url.pathname ===
                "/admin" &&
            request.method === "GET"
        ) {

            const authenticated =
                await requireAdmin(
                    request,
                    env
                );

            if (!authenticated) {

                return new Response(
                    LOGIN_HTML,
                    {
                        status: 401,

                        headers: {
                            "Content-Type":
                                "text/html;charset=UTF-8",

                            "Cache-Control":
                                "no-store"
                        }
                    }
                );
            }

            return new Response(
                ADMIN_HTML,
                {
                    status: 200,

                    headers: {
                        "Content-Type":
                            "text/html;charset=UTF-8",

                        "Cache-Control":
                            "private, no-store"
                    }
                }
            );
        }


        // =================================================
        // PUBLICO /
        // =================================================

        if (
            url.pathname === "/" &&
            request.method === "GET"
        ) {

            return new Response(
                PUBLIC_HTML,
                {
                    status: 200,

                    headers: {
                        "Content-Type":
                            "text/html;charset=UTF-8",

                        "Cache-Control":
                            "no-store"
                    }
                }
            );
        }


        // =================================================
        // PUBLICO /files
        // =================================================

        if (
            url.pathname === "/files" &&
            request.method === "GET"
        ) {

            return handleFiles(
                env
            );
        }


        // =================================================
        // PUBLICO /f/...
        // =================================================

        if (
            url.pathname.startsWith("/f/") &&
            request.method === "GET"
        ) {

            return handleFile(
                request,
                env
            );
        }


        // =================================================
        // DESDE AQUI: SOLO ADMIN
        // =================================================

        const authenticated =
            await requireAdmin(
                request,
                env
            );

        if (!authenticated) {

            return new Response(
                "No autorizado.",
                {
                    status: 401
                }
            );
        }


        // =================================================
        // UPLOAD
        // =================================================

        if (
            url.pathname === "/upload" &&
            request.method === "POST"
        ) {

            return handleUpload(
                request,
                env
            );
        }


        // =================================================
        // DELETE
        // =================================================

        if (
            url.pathname.startsWith("/f/") &&
            request.method === "DELETE"
        ) {

            return handleDeleteFile(
                request,
                env
            );
        }


        // =================================================
        // EDIT GET
        // =================================================

        if (
            url.pathname.startsWith("/edit/") &&
            request.method === "GET"
        ) {

            return handleEdit(
                request,
                env
            );
        }


        // =================================================
        // EDIT PUT
        // =================================================

        if (
            url.pathname.startsWith("/edit/") &&
            request.method === "PUT"
        ) {

            return handleSaveEdit(
                request,
                env
            );
        }


        // =================================================
        // 404
        // =================================================

        return new Response(
            "No encontrado.",
            {
                status: 404
            }
        );
    }
};
